-- FARDO CONSULTING — сохтори пойгоҳи маълумот (SQLite)
--
-- МУҲИМ (талаботи бехатарӣ): дар ин пойгоҳ ҲЕҶ ГОҲ сутуне барои
-- логин/пароли андоз, омор ё ФСЗН илова накунед. Боти мо танҳо
-- маълумоти хидматрасонии муҳосибиро нигоҳ медорад, на рамзҳои
-- воридшавӣ ба системаҳои давлатӣ.

PRAGMA foreign_keys = ON;

-- ---------------------------------------------------------------------
-- Корхонаҳо
-- ---------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS companies (
    id          INTEGER PRIMARY KEY AUTOINCREMENT,
    inn         TEXT NOT NULL UNIQUE,      -- РМА, ҳамчун матн (то 0-ҳои пешин гум нашаванд)
    name        TEXT NOT NULL,
    director    TEXT,
    accountant  TEXT,
    tax_office  TEXT,
    reg_date    TEXT,                      -- санаи бақайдгирӣ (матн, формати DD.MM.YYYY)
    created_at  TEXT NOT NULL DEFAULT (datetime('now')),
    updated_at  TEXT NOT NULL DEFAULT (datetime('now'))
);

-- Феҳристи телефонҳои иҷозатдодашуда барои ҳар корхона (як-ба-бисёр).
-- Санҷиш аз рӯи 9 рақами охирин сурат мегирад (phone_last9).
CREATE TABLE IF NOT EXISTS company_phones (
    id          INTEGER PRIMARY KEY AUTOINCREMENT,
    company_id  INTEGER NOT NULL REFERENCES companies(id) ON DELETE CASCADE,
    phone_last9 TEXT NOT NULL,             -- 9 рақами охирин, бе +992/фосила
    label       TEXT,                      -- ихтиёрӣ: "Офис", "Зулфиқор" ва ғ.
    created_at  TEXT NOT NULL DEFAULT (datetime('now'))
);
CREATE INDEX IF NOT EXISTS idx_company_phones_phone ON company_phones(phone_last9);
CREATE INDEX IF NOT EXISTS idx_company_phones_company ON company_phones(company_id);

-- ---------------------------------------------------------------------
-- Корбарон (миҷозони воридшуда ба бот)
-- ---------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS users (
    id              INTEGER PRIMARY KEY AUTOINCREMENT,
    tg_user_id      INTEGER NOT NULL UNIQUE,
    company_id      INTEGER NOT NULL REFERENCES companies(id) ON DELETE CASCADE,
    verified_phone  TEXT,                  -- 9 рақами тасдиқшуда
    created_at      TEXT NOT NULL DEFAULT (datetime('now'))
);

-- Назорати кӯшишҳои нодурусти РМА — 5 нодуруст → қуфли 15 дақиқа.
CREATE TABLE IF NOT EXISTS login_lockouts (
    tg_user_id      INTEGER PRIMARY KEY,
    failed_count    INTEGER NOT NULL DEFAULT 0,
    locked_until    TEXT,                  -- NULL = қуфл нест
    last_attempt_at TEXT NOT NULL DEFAULT (datetime('now'))
);

-- Дархостҳои вуруд, вақте ки телефон дар феҳрист нест → ба админ меравад.
CREATE TABLE IF NOT EXISTS access_requests (
    id              INTEGER PRIMARY KEY AUTOINCREMENT,
    tg_user_id      INTEGER NOT NULL,
    company_id      INTEGER NOT NULL REFERENCES companies(id) ON DELETE CASCADE,
    phone_submitted TEXT NOT NULL,
    status          TEXT NOT NULL DEFAULT 'pending'
                        CHECK (status IN ('pending', 'approved', 'denied')),
    requested_at    TEXT NOT NULL DEFAULT (datetime('now')),
    decided_at      TEXT,
    decided_by_tg   INTEGER                -- ID-и админи қарордиҳанда
);
CREATE INDEX IF NOT EXISTS idx_access_requests_status ON access_requests(status);

-- ---------------------------------------------------------------------
-- 📑 Ҳисоботҳо
-- ---------------------------------------------------------------------
-- Мақом:
--   'submitted'  — админ дар Экзел ҳамчун супоридашуда қайд кардааст
--   'in_review'  — админ дар Экзел ҳамчун дар тафтиш қайд кардааст
--   NULL         — ҳоло муайян карда нашудааст: бот худаш аз рӯи
--                  due_date ҳисоб мекунад (⚠️ ≤7 рӯз, ❌ гузашта,
--                  вагарна 🕓 дар интизор)
CREATE TABLE IF NOT EXISTS reports (
    id          INTEGER PRIMARY KEY AUTOINCREMENT,
    company_id  INTEGER NOT NULL REFERENCES companies(id) ON DELETE CASCADE,
    report_type TEXT NOT NULL,             -- масалан: "Фоида", "ПНВ/НДС", "Омор", "ФСЗН"
    due_date    TEXT NOT NULL,             -- формати YYYY-MM-DD, барои муқоисаи сана
    status      TEXT CHECK (status IN ('submitted', 'in_review') OR status IS NULL),
    updated_at  TEXT NOT NULL DEFAULT (datetime('now'))
);
CREATE INDEX IF NOT EXISTS idx_reports_company ON reports(company_id);
CREATE INDEX IF NOT EXISTS idx_reports_due_date ON reports(due_date);

-- ---------------------------------------------------------------------
-- 💰 Қарздорӣ (як сабт барои ҳар корхона, воридкунӣ = upsert)
-- ---------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS debts (
    company_id  INTEGER PRIMARY KEY REFERENCES companies(id) ON DELETE CASCADE,
    amount      NUMERIC NOT NULL DEFAULT 0,
    updated_at  TEXT NOT NULL DEFAULT (datetime('now'))
);

-- ---------------------------------------------------------------------
-- 🏦 Ҳисобҳои бонкӣ
-- ---------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS bank_accounts (
    id              INTEGER PRIMARY KEY AUTOINCREMENT,
    company_id      INTEGER NOT NULL REFERENCES companies(id) ON DELETE CASCADE,
    bank_name       TEXT NOT NULL,
    location        TEXT,
    account_number  TEXT NOT NULL
);
CREATE INDEX IF NOT EXISTS idx_bank_accounts_company ON bank_accounts(company_id);

-- ---------------------------------------------------------------------
-- 📝 Шартнома ва пардохт
-- ---------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS contracts (
    company_id  INTEGER PRIMARY KEY REFERENCES companies(id) ON DELETE CASCADE,
    start_date  TEXT,                      -- YYYY-MM-DD
    end_date    TEXT,                      -- YYYY-MM-DD; огоҳӣ агар ≤30 рӯз монда бошад
    monthly_fee NUMERIC NOT NULL DEFAULT 0
);

CREATE TABLE IF NOT EXISTS payments (
    id              INTEGER PRIMARY KEY AUTOINCREMENT,
    company_id      INTEGER NOT NULL REFERENCES companies(id) ON DELETE CASCADE,
    payment_date    TEXT NOT NULL,         -- YYYY-MM-DD
    amount          NUMERIC NOT NULL,
    note            TEXT
);
CREATE INDEX IF NOT EXISTS idx_payments_company_date ON payments(company_id, payment_date);

-- ---------------------------------------------------------------------
-- ❓ Саволҳо
-- ---------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS questions (
    id              INTEGER PRIMARY KEY AUTOINCREMENT,
    tg_user_id      INTEGER NOT NULL,
    company_id      INTEGER NOT NULL REFERENCES companies(id) ON DELETE CASCADE,
    question_text   TEXT NOT NULL,
    asked_at        TEXT NOT NULL DEFAULT (datetime('now')),
    answer_text     TEXT,
    answered_at     TEXT,
    answered_by_tg  INTEGER
);
CREATE INDEX IF NOT EXISTS idx_questions_company ON questions(company_id);
CREATE INDEX IF NOT EXISTS idx_questions_user ON questions(tg_user_id);

-- ---------------------------------------------------------------------
-- 📄 Маълумотномаҳо
-- ---------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS certificates (
    id                  INTEGER PRIMARY KEY AUTOINCREMENT,
    tg_user_id          INTEGER NOT NULL,
    company_id          INTEGER NOT NULL REFERENCES companies(id) ON DELETE CASCADE,
    cert_type           TEXT NOT NULL CHECK (cert_type IN ('no_tax_debt', 'accounting', 'other')),
    status              TEXT NOT NULL DEFAULT 'in_progress'
                            CHECK (status IN ('in_progress', 'ready')),
    requested_at        TEXT NOT NULL DEFAULT (datetime('now')),
    file_telegram_id    TEXT,               -- file_id-и Телеграм, пас аз тайёр шудан
    delivered_at        TEXT
);
CREATE INDEX IF NOT EXISTS idx_certificates_company ON certificates(company_id);
