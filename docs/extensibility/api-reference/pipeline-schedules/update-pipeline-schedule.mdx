---
title: "Update schedule"
api: "PUT /api/pipeline_schedules/:id"
---

<ParamField path="id" type="integer" required>
  Pipeline schedule ID
</ParamField>

<b>Body</b>
<br />
<br />

<ParamField body="pipeline_schedule" type="object" required>
  <Expandable title="payload" defaultOpen="true">
    <ParamField body="event_matchers" type="array of objects">
      Deprecated. Used only for maintaining existing legacy event triggers.
    </ParamField>
    <ParamField body="name" type="string">
      Trigger names are trimmed before validation. The name must include at least one letter or
      number, cannot exceed 255 characters, and must be unique within the pipeline.
    </ParamField>
    <ParamField body="schedule_interval" type="string">
      Required when `schedule_type` is `time`.
    </ParamField>
    <ParamField body="schedule_type" type="string">
      Trigger type. Use `api` or `time` for supported non-legacy triggers. `event` is deprecated
      and only used for existing legacy event triggers.
    </ParamField>
    <ParamField body="settings" type="object">
      Additional settings for the trigger.

      <Expandable title="properties">
        <ParamField body="timeout" type="integer">
          Optional run timeout in seconds. When provided, the value must be a positive integer.
          Use `null` or omit this field for no timeout.
        </ParamField>
      </Expandable>
    </ParamField>
    <ParamField body="sla" type="integer">
      Optional SLA threshold in seconds. When provided, the value must be a positive integer.
      Use `0` or `null` to clear an existing SLA, or omit this field for no SLA change.
    </ParamField>
    <ParamField body="start_time" type="datetime">
      Persisted schedule start timestamp. For `@once`, this value determines the single run time;
      for recurring time schedules, it anchors when the schedule begins.
    </ParamField>
    <ParamField body="status" type="string">
    </ParamField>
    <ParamField body="tags" type="array of strings">
      Non-empty tag names to associate with the trigger. Tag names are trimmed and duplicate names
      are ignored before persistence. Each tag name cannot exceed 70 characters.
    </ParamField>
    <ParamField body="variables" type="object">
    </ParamField>
  </Expandable>
</ParamField>
